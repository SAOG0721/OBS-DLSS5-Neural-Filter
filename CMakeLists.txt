cmake_minimum_required(VERSION 3.25)
project(Dlss5NeuralFilter VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "DLSS5 Neural Filter needs Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(OBS_SOURCE_ROOT "" CACHE PATH "OBS headers; build baseline only, no runtime version restriction")
set(NGX_SDK_ROOT "" CACHE PATH "External NVIDIA DLSS SDK")
set(FIDELITYFX_SDK_ROOT "" CACHE PATH "External FidelityFX v2.3.0 SDK")
set(NVOF_SDK_ROOT "" CACHE PATH "External NVIDIA D3D12 optical-flow API headers")
set(DXC_PATH "" CACHE FILEPATH "DXC compiler for FidelityFX shaders")
set(FFX_ROOT "${FIDELITYFX_SDK_ROOT}/Kits/FidelityFX")
foreach(required IN ITEMS "${OBS_SOURCE_ROOT}/libobs/obs.h" "${NGX_SDK_ROOT}/include/nvsdk_ngx.h"
    "${NGX_SDK_ROOT}/lib/Windows_x86_64/x64/nvsdk_ngx_s.lib" "${FFX_ROOT}/backend/dx12/ffx_dx12.cpp"
    "${NVOF_SDK_ROOT}/nvOpticalFlowD3D12.h" "${DXC_PATH}")
    if(NOT EXISTS "${required}")
        message(FATAL_ERROR "Dependency missing: ${required}")
    endif()
endforeach()
set(FFX_SHADERS "${CMAKE_BINARY_DIR}/FidelityFXOpticalFlow")
add_custom_target(GenerateOpticalFlowShaders
    COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/scripts/Generate-FidelityFXOpticalFlowShaders.ps1"
        -SdkDirectory "${FIDELITYFX_SDK_ROOT}" -OutputDirectory "${FFX_SHADERS}" -DxcPath "${DXC_PATH}"
    VERBATIM)
add_library(neural_flow STATIC src/FlowGpu.cpp
    "${FFX_ROOT}/api/internal/ffx_assert.cpp"
    "${FFX_ROOT}/api/internal/ffx_message.cpp"
    "${FFX_ROOT}/api/internal/ffx_object_management.cpp"
    "${FFX_ROOT}/backend/dx12/ffx_dx12.cpp"
    "${FFX_ROOT}/backend/dx12/ffx_backends_dx12.cpp"
    "${FFX_ROOT}/framegeneration/fsr3/internal/ffx_opticalflow.cpp"
    "${FFX_ROOT}/framegeneration/fsr3/internal/ffx_opticalflow_shaderblobs.cpp")
add_dependencies(neural_flow GenerateOpticalFlowShaders)
target_include_directories(neural_flow PRIVATE "${FFX_SHADERS}" "${NVOF_SDK_ROOT}"
    "${FFX_ROOT}/backend/dx12" "${FFX_ROOT}/framegeneration/fsr3/include" src/compat)
target_link_libraries(neural_flow PUBLIC d3d12 dxgi)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/data/neural.hlsl" NEURAL_SHADER_SOURCE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS data/neural.hlsl)
configure_file(cmake/ShaderSource.h.in "${CMAKE_BINARY_DIR}/generated/ShaderSource.h" @ONLY)
file(WRITE "${CMAKE_BINARY_DIR}/generated/obsconfig.h"
    "#pragma once\n#define OBS_DATA_PATH \"data/obs-studio\"\n#define OBS_INSTALL_PREFIX \"\"\n#define OBS_RELEASE_CANDIDATE 0\n#define OBS_BETA 0\n")
add_library(neural_backend STATIC src/NgxSession.cpp src/GpuBackend.cpp)
target_include_directories(neural_backend PUBLIC src PRIVATE "${CMAKE_BINARY_DIR}/generated" "${NGX_SDK_ROOT}/include")
target_link_libraries(neural_backend PUBLIC neural_flow d3d11 d3d12 dxgi d3dcompiler bcrypt version wintrust
    "${NGX_SDK_ROOT}/lib/Windows_x86_64/x64/nvsdk_ngx_s.lib")
add_library(obs-dlss5 MODULE src/ObsApi.cpp src/Filter.cpp)
target_include_directories(obs-dlss5 PRIVATE src "${CMAKE_BINARY_DIR}/generated")
target_include_directories(obs-dlss5 SYSTEM PRIVATE "${OBS_SOURCE_ROOT}/libobs")
target_link_libraries(obs-dlss5 PRIVATE neural_backend)
target_compile_definitions(obs-dlss5 PRIVATE NEURAL_PLUGIN_VERSION="${PROJECT_VERSION}")
set_target_properties(obs-dlss5 PROPERTIES PREFIX "")
foreach(target IN ITEMS neural_flow neural_backend obs-dlss5)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${target} PRIVATE /utf-8 /EHsc)
endforeach()
target_compile_options(neural_backend PRIVATE /W4 /permissive-)
target_compile_options(obs-dlss5 PRIVATE /W4 /permissive-)
add_custom_command(TARGET obs-dlss5 POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_BINARY_DIR}/package/obs-dlss5/bin/64bit"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:obs-dlss5>" "${CMAKE_BINARY_DIR}/package/obs-dlss5/bin/64bit/"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/data" "${CMAKE_BINARY_DIR}/package/obs-dlss5/data"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/licenses" "${CMAKE_BINARY_DIR}/package/obs-dlss5/data/licenses"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" "${CMAKE_BINARY_DIR}/package/obs-dlss5/data/licenses/GPL-3.0.txt"
    VERBATIM)
